1. 在 repository 里增加一个取最近 N 根 K 线的函数

在 data/repository.py 里新增，例如：

def get_recent_kline(
    ts_code: str,
    freq: str,
    count: int,
) -> pd.DataFrame:
    """
    从 stock_kline 中取某只股票、某个 freq 的最近 count 根 K 线，
    按 datetime 升序返回。没有数据则返回空 DataFrame。
    """
    eng = get_engine()
    _ensure_stock_kline(eng)
    from sqlalchemy import text as _text
    with eng.connect() as conn:
        conn.rollback()
        conn = conn.execution_options(isolation_level="AUTOCOMMIT")
        sql = _text(
            """
            select ts_code, freq, datetime, open, high, low, close, volume, amount
            from stock_kline
            where ts_code = :ts and freq = :fq
            order by datetime desc
            limit :n
            """
        )
        df = pd.read_sql(
            sql,
            conn,
            params={"ts": ts_code, "fq": freq, "n": count},
            parse_dates=["datetime"],
        )
    if df.empty:
        return df
    return df.sort_values("datetime").reset_index(drop=True)


要求：

不要改已有的写入逻辑（upsert_stock_kline 等），只新增查询函数；

freq 的值与现有逻辑保持一致：日线 "1d"，分钟 "60m" / "30m" / "15m" / "5m" 等。

2. 新增 HybridDataSource 实现

新建文件：data/hybrid_source.py，内容示例：

from __future__ import annotations

from datetime import date, datetime
from typing import Optional

import pandas as pd

from .base_source import DataSource
from .pytdx_source import PytdxDataSource
from . import repository


class HybridDataSource(DataSource):
    """
    DB 优先的数据源：
    - get_daily_bars / get_minute_bars：
      先从 stock_kline 取最近 N 根；
      如果数量不足，再从 PytdxDataSource 拉取，写回 stock_kline，再从 DB 统一取。
    - get_ticks：直接走 PytdxDataSource（暂不落库）。
    """

    def __init__(self, inner: Optional[PytdxDataSource] = None) -> None:
        self._inner = inner or PytdxDataSource()

    # --- 日线 ---

    def get_daily_bars(self, ts_code: str, count: int = 240) -> pd.DataFrame:
        freq = "1d"
        df_db = repository.get_recent_kline(ts_code, freq, count)
        if len(df_db) >= count:
            return df_db

        # 不足部分用 pytdx 补
        df_remote = self._inner.get_daily_bars(ts_code, count=count)
        if not df_remote.empty:
            repository.upsert_stock_kline(ts_code, freq, df_remote)
            df_db = repository.get_recent_kline(ts_code, freq, count)
            if not df_db.empty:
                return df_db

        # 兜底：至少返回远端数据
        return df_remote

    # --- 分钟线 ---

    def get_minute_bars(
        self,
        ts_code: str,
        freq: str = "1m",
        count: int = 240,
    ) -> pd.DataFrame:
        # freq 直接作为 stock_kline.freq 使用（例如 "60m"、"30m"）
        df_db = repository.get_recent_kline(ts_code, freq, count)
        if len(df_db) >= count:
            return df_db

        df_remote = self._inner.get_minute_bars(ts_code, freq=freq, count=count)
        if not df_remote.empty:
            repository.upsert_stock_kline(ts_code, freq, df_remote)
            df_db = repository.get_recent_kline(ts_code, freq, count)
            if not df_db.empty:
                return df_db

        return df_remote

    # --- tick 直接透传 ---

    def get_ticks(
        self,
        ts_code: str,
        trade_date: Optional[date] = None,
        count: int = 2000,
    ) -> pd.DataFrame:
        return self._inner.get_ticks(ts_code, trade_date=trade_date, count=count)

    # 可选：如果上层需要 get_kline，可以简单复用 inner 的逻辑，
    # 或根据需要再加一层 DB 优先封装。


要求：

HybridDataSource 必须继承 DataSource 并实现 get_daily_bars / get_minute_bars / get_ticks 三个方法；

不要删除/改动 PytdxDataSource 原有逻辑，Hybrid 只是一个上层包装；

对于 UI 场景，不需要额外实现更复杂的接口（回测/批处理仍可用原有 PytdxDataSource）。
